#   python tools/fetch_mlb_logos.py          # cache raw SVG/PNG art (stdlib only)
#   python tools/render_mlb_logo_variants.py # composite the 20x20 base + alt tiles
#   python tools/fetch_nfl_logos.py          # the same for the NFL
#   python tools/render_nfl_logo_variants.py
#

cairosvg==2.9.0
scipy==1.18.0
